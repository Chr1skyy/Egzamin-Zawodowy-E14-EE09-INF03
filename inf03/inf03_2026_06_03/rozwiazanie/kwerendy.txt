1. SELECT nazwa, styl, cena FROM meble WHERE czyNowy = 1 AND cena <= 1000 ORDER BY cena;
2. CREATE VIEW komody AS SELECT idMeble, nazwa, kategoria, cena FROM meble WHERE kategoria = 3;
3. SELECT ROUND(AVG(cena), 0) AS 'Średnia cena komód', MIN(cena) AS 'Minimalna', MAX(cena) AS 'Maksymalna' FROM komody;
